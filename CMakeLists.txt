cmake_minimum_required(VERSION 3.20)
project(conspiracy_board LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(conspiracy_core src/graph_engine.cpp)
target_include_directories(conspiracy_core PUBLIC src)
if(EMSCRIPTEN)
  add_executable(engine src/bindings.cpp)
  target_link_libraries(engine PRIVATE conspiracy_core)
  target_link_options(engine PRIVATE "--bind" "-sMODULARIZE=1" "-sEXPORT_ES6=1" "-sEXPORT_NAME=createConspiracyEngine" "-sALLOW_MEMORY_GROWTH=1" "-sENVIRONMENT=web" "-O3")
  set_target_properties(engine PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/web" SUFFIX ".js")
else()
  enable_testing()
  add_executable(engine_tests tests/engine_test.cpp)
  target_link_libraries(engine_tests PRIVATE conspiracy_core)
  add_test(NAME graph_engine_tests COMMAND engine_tests)
endif()
